cmake_minimum_required(VERSION 3.25)
project(BShareVirtualCamera VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "BShare virtual camera currently requires Windows 11 x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
set(BSHARE_WIL_SOURCE_DIR "" CACHE PATH "Optional local checkout of pinned Microsoft WIL")
if(BSHARE_WIL_SOURCE_DIR)
  set(wil_SOURCE_DIR "${BSHARE_WIL_SOURCE_DIR}")
else()
  FetchContent_Declare(wil GIT_REPOSITORY https://github.com/microsoft/wil.git
    GIT_TAG a1c0cbe1d7065aef6333399d3e25608eacdffc0f GIT_SHALLOW FALSE)
  FetchContent_GetProperties(wil)
  if(NOT wil_POPULATED)
    FetchContent_Populate(wil)
  endif()
endif()
find_path(CPPWINRT_INCLUDE_DIR winrt/base.h
  PATHS "C:/Program Files (x86)/Windows Kits/10/Include/${CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION}/cppwinrt"
  REQUIRED)
add_library(bshare-virtual-camera SHARED
  src/dllmain.cpp src/exports.def src/winrtCommon.cpp
  src/SimpleMediaSource.cpp src/SimpleMediaStream.cpp
  src/VirtualCameraMediaSourceActivate.cpp src/SimpleFrameGenerator.cpp)
target_include_directories(bshare-virtual-camera PRIVATE src include "${wil_SOURCE_DIR}/include" "${CPPWINRT_INCLUDE_DIR}")
target_compile_definitions(bshare-virtual-camera PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(bshare-virtual-camera PRIVATE /W4 /EHsc /utf-8 /permissive-)
target_link_libraries(bshare-virtual-camera PRIVATE mf mfplat mfuuid mfsensorgroup ole32 runtimeobject windowsapp advapi32 gdi32)
add_executable(bshare-camera src/camera_helper.cpp)
target_include_directories(bshare-camera PRIVATE src include)
target_compile_definitions(bshare-camera PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(bshare-camera PRIVATE /W4 /EHsc /utf-8 /permissive-)
target_link_libraries(bshare-camera PRIVATE mf mfplat mfreadwrite mfuuid mfsensorgroup ole32 advapi32)
set_property(TARGET bshare-virtual-camera bshare-camera PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BSHARE_CAMERA_INSTALL_DIR ".")
if(TARGET BShare)
  # Flutter stores a generator expression in CMAKE_INSTALL_PREFIX. Absolute
  # destinations evaluate it correctly; prefix-relative paths do not.
  set(BSHARE_CAMERA_INSTALL_DIR "$<TARGET_FILE_DIR:BShare>")
endif()
install(TARGETS bshare-virtual-camera bshare-camera RUNTIME DESTINATION "${BSHARE_CAMERA_INSTALL_DIR}")
install(FILES README.md MICROSOFT-LICENSE DESTINATION "${BSHARE_CAMERA_INSTALL_DIR}/data/virtual-camera")
option(BSHARE_VIRTUAL_CAMERA_TESTS "Build virtual camera transport checks" ${PROJECT_IS_TOP_LEVEL})
if(BSHARE_VIRTUAL_CAMERA_TESTS)
  enable_testing()
  add_executable(bshare-camera-transport-test tests/transport_test.cpp src/SimpleFrameGenerator.cpp)
  target_include_directories(bshare-camera-transport-test PRIVATE src include "${wil_SOURCE_DIR}/include" "${CPPWINRT_INCLUDE_DIR}")
  target_compile_definitions(bshare-camera-transport-test PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(bshare-camera-transport-test PRIVATE /EHsc /utf-8)
  target_link_libraries(bshare-camera-transport-test PRIVATE mf mfplat mfuuid mfsensorgroup ole32 runtimeobject windowsapp advapi32 gdi32)
  set_property(TARGET bshare-camera-transport-test PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_test(NAME virtual-camera-transport COMMAND bshare-camera-transport-test)
  set_tests_properties(virtual-camera-transport PROPERTIES TIMEOUT 30)
  add_test(NAME virtual-camera-source-lifecycle COMMAND bshare-camera --probe-direct-cycle "$<TARGET_FILE:bshare-virtual-camera>" 5)
  set_tests_properties(virtual-camera-source-lifecycle PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE)
endif()
