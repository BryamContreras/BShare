cmake_minimum_required(VERSION 3.16)
project(BShareLinuxVirtualCamera LANGUAGES CXX)
find_package(Threads REQUIRED)
add_executable(bshare-linux-camera camera_helper_linux.cpp)
set_target_properties(bshare-linux-camera PROPERTIES OUTPUT_NAME "bshare-camera" CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES)
target_compile_definitions(bshare-linux-camera PRIVATE _GNU_SOURCE)
target_compile_options(bshare-linux-camera PRIVATE -Wall -Wextra -Werror)
target_link_libraries(bshare-linux-camera PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
option(BSHARE_LINUX_CAMERA_TESTS "Build Linux virtual-camera IPC/device contracts" OFF)
if(BSHARE_LINUX_CAMERA_TESTS)
  enable_testing()
  add_executable(bshare-linux-camera-test transport_test.cpp)
  target_compile_definitions(bshare-linux-camera-test PRIVATE _GNU_SOURCE)
  target_compile_options(bshare-linux-camera-test PRIVATE -Wall -Wextra -Werror)
  target_link_libraries(bshare-linux-camera-test PRIVATE Threads::Threads)
  add_test(NAME linux-virtual-camera-transport COMMAND bshare-linux-camera-test)
  set_tests_properties(linux-virtual-camera-transport PROPERTIES TIMEOUT 20)
endif()
